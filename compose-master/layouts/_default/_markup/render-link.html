{{- /* Markdown links. 1. Bare URLs auto-linked by goldmark linkify that are placeholders (example.com, localhost, IP addresses, {address}, xx) are printed as plain text — they lead nowhere. 2. Links whose text says nothing on its own ("More info", "here", … — i18n key generic-link-texts) get a hidden continuation for screen readers / SEO audits: More info: Software Stack Versions. Source: heading of the anchor on the same page → title of the target docs page → site.Params.linkLabels (URL prefix → label) for other sites → nothing. */ -}} {{- $link := .Destination }}{{ $isRemote := strings.HasPrefix $link "http" }}{{ $isDocs := in $link .Page.Site.Params.hostName }}{{ $isVirtuozzoSites := in $link "virtuozzo.com" }}{{ $isPrivate := in $link "/private-" -}} {{- $plain := .PlainText | htmlUnescape | replaceRE `^\s+|\s+$` "" -}} {{- $isAuto := and (ne $plain "") (or (eq $plain $link) (eq (printf "http://%s" $plain) $link) (eq (printf "https://%s" $plain) $link)) -}} {{- $isPlaceholder := false -}} {{- if $isAuto -}} {{- $host := $link | replaceRE `^[a-zA-Z]+://([^/?#]*).*$` "$1" | lower -}} {{- $isPlaceholder = or (gt (len (findRE `(^|\.)example\.(com|org|net)(:|$)|^localhost(:|$)|^\d{1,3}(\.\d{1,3}){3}(:|$)|x{2}|[{}<>*]` $host 1)) 0) (gt (len (findRE `[{}<>]|%7[bBdD]` $link 1)) 0) -}} {{- end -}} {{- $sr := "" -}} {{- $norm := $plain | lower | replaceRE `\s+` " " | replaceRE `[\s.:!?»›>→\-–—]+$` "" | replaceRE `^\s+|\s+$` "" -}} {{- if and (not $isAuto) (ne $norm "") -}} {{- $generic := split (partial "i18n-extended.html" (slice "generic-link-texts" "")) "|" -}} {{- if in $generic $norm -}} {{- if and $isRemote (not $isDocs) -}} {{- /* other sites: URL prefix → label; site.Params.linkLabels adds to / overrides the defaults */ -}} {{- $labels := dict "https://www.virtuozzo.com/application-management-api-docs" "Virtuozzo Application Management API" "https://docs.cloudscripting.com" "CloudScripting documentation" -}} {{- with site.Params.linkLabels }}{{ $labels = merge $labels . }}{{ end -}} {{- range $prefix, $label := $labels -}} {{- if strings.HasPrefix (lower $link) (lower $prefix) }}{{ $sr = $label }}{{ end -}} {{- end -}} {{- else -}} {{- $u := urls.Parse $link -}} {{- $target := false -}} {{- if eq $u.Path "" -}} {{- if $u.Fragment }}{{ $target = $.Page }}{{ end -}} {{- else -}} {{- $m := partialCached "link-target-map.html" . site.Language.Lang -}} {{- $base := (urls.Parse site.BaseURL).Path -}} {{- $keys := slice $u.Path (printf "%s/" (path.Join $base $u.Path)) (printf "%s/" (path.Join $.Page.RelPermalink $u.Path)) -}} {{- range $keys -}}{{- if not $target -}}{{- with index $m . -}}{{- $target = . -}}{{- end -}}{{- end -}}{{- end -}} {{- end -}} {{- with $target -}} {{- $sr = .LinkTitle -}} {{- if and $u.Fragment (eq . $.Page) -}} {{- with index $.Page.Fragments.HeadingsMap $u.Fragment }}{{ $sr = .Title | plainify | htmlUnescape }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- if and $sr (in (lower $plain) (lower $sr)) }}{{ $sr = "" }}{{ end -}} {{- end -}} {{- end -}} {{- if $isPlaceholder -}} {{ .Text | safeHTML }} {{- else -}} {{ .Text | safeHTML }}{{ with $sr }}: {{ . }}{{ end }} {{- end -}}
