{{- /* One level of the menu in the page. Args: dict "menu" (sidebar/menu.html) "ids" "level" "open" (ids opened on this page: the current page's ancestors and the current page) "current" (ids of the current page's items) "depth" (levels always in the page). In the page: the first "depth" levels of the whole menu (params.sidebarDepth, 3 by default), collapsed unless open, so that every page links to the top of the docs in a couple of clicks, as the full menu did for search engines; and the open branch of the current page down to its children. Deeper children of other items are not in the page: such an item has data-id, and sidebar.js renders them from menu.json when it is opened. */ -}}
    {{- range .ids -}} {{- $n := index $.menu.nodes . -}} {{- $open := and $n.kids (in $.open .) -}} {{- $render := and $n.kids (or $open (lt $.level $.depth)) -}} {{- $cls := $n.cls -}} {{- if in $.current . }}{{ $cls = printf "%s current-page" $cls }}{{ end -}} {{- if $open }}{{ $cls = printf "%s expanded" $cls }}{{ end }} {{- $n.html | safeHTML -}} {{- if $render }}{{ partial "sidebar/branch.html" (dict "menu" $.menu "ids" $n.kids "level" (add $.level 1) "open" $.open "current" $.current "depth" $.depth) }}{{ end -}} {{- end }}
