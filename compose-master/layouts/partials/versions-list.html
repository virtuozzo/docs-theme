{{- /* Version switcher links. Docs root = baseURL without the trailing //. params.defaultVersion (passed by the deploy workflow as HUGO_PARAMS_DEFAULTVERSION) lives at the docs root, every other version at /. Without defaultVersion (local hugo server) the current version links to the current build's home. */ -}} {{- $cur := print site.Params.version -}} {{- $def := "" -}}{{- with site.Params.defaultVersion -}}{{- $def = print . -}}{{- end -}} {{- $root := replaceRE (printf `%s/$` (replace $cur "." `\.`)) "" site.BaseURL -}} {{- /* params.versions: a list from the branch config, or a comma-separated string from the deploy pipeline (HUGO_PARAMS_VERSIONS, read from the main branch config; sent with a trailing comma so a single version stays a string and "9.0" is not read as the number 9) */ -}} {{- $versions := site.Params.versions -}} {{- if and $versions (not (reflect.IsSlice $versions)) -}} {{- $list := slice -}} {{- range split (print $versions) "," -}}{{- with trim . " " -}}{{- $list = $list | append . -}}{{- end -}}{{- end -}} {{- $versions = $list -}} {{- end -}} {{- range $versions -}} {{- $v := print . -}} {{- $href := print $root $v "/" -}} {{- if eq $v $def -}}{{- $href = $root -}} {{- else if and (not $def) (eq $v $cur) -}}{{- $href = site.Home.Permalink -}} {{- end }} {{ $v }} {{- end -}}
